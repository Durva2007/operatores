#include <iostream>
using namespace std;

int main()
{
    int arr[100], n, i, pos, value, key;
    bool found = false;

    // Create Array
    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter array elements: ";
    for(i = 0; i < n; i++)
        cin >> arr[i];

    // Traversal
    cout << "\nArray elements are: ";
    for(i = 0; i < n; i++)
        cout << arr[i] << " ";

    // Insertion
    cout << "\n\nEnter position for insertion: ";
    cin >> pos;
    cout << "Enter element to insert: ";
    cin >> value;

    for(i = n; i >= pos; i--)
        arr[i] = arr[i - 1];

    arr[pos - 1] = value;
    n++;

    cout << "\nArray after insertion: ";
    for(i = 0; i < n; i++)
        cout << arr[i] << " ";

    // Deletion
    cout << "\n\nEnter position for deletion: ";
    cin >> pos;

    for(i = pos - 1; i < n - 1; i++)
        arr[i] = arr[i + 1];

    n--;

    cout << "\nArray after deletion: ";
    for(i = 0; i < n; i++)
        cout << arr[i] << " ";

    // Search
    cout << "\n\nEnter element to search: ";
    cin >> key;

    for(i = 0; i < n; i++)
    {
        if(arr[i] == key)
        {
            found = true;
            cout << "Element found at position " << i + 1;
            break;
        }
    }

    if(!found)
        cout << "Element not found";

    return 0;
}
